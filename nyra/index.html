const messagesEl = document.getElementById('messages'); const input = document.getElementById('prompt'); const chatForm = document.getElementById('chatForm'); const sendButton = chatForm.querySelector('button[type="submit"]'); // Current-tab session history; survives page refresh. const STORAGE_KEY = 'nyra_session_history_v1'; const MAX_HISTORY = 30; let history = []; try { const saved = JSON.parse(sessionStorage.getItem(STORAGE_KEY) || '[]'); if (Array.isArray(saved)) { history = saved.filter(item => item && ['user', 'assistant'].includes(item.role) && typeof item.content === 'string' ).slice(-MAX_HISTORY); } } catch { history = []; } function saveHistory() { try { sessionStorage.setItem( STORAGE_KEY, JSON.stringify(history.slice(-MAX_HISTORY)) ); } catch (error) { console.warn('Nyra session history could not be saved.'); } } function renderHistory() { messagesEl.replaceChildren(); const welcome = document.createElement('div'); welcome.className = 'message nyra'; welcome.textContent = "NYRA\nHello! I'm Nyra, your AI companion from AstropicAI."; messagesEl.appendChild(welcome); history.forEach(item => { addMessage( (item.role === 'user' ? 'You: ' : 'Nyra: ') + item.content, item.role === 'user' ? 'user' : 'nyra' ); }); } function addMessage(text, kind) { const el = document.createElement('div'); el.className = 'message ' + kind; el.textContent = text; messagesEl.appendChild(el); messagesEl.scrollTop = messagesEl.scrollHeight; } renderHistory(); document.querySelectorAll('[data-prompt]').forEach(button => { button.addEventListener('click', () => { input.value = button.dataset.prompt; input.focus(); }); }); chatForm.addEventListener('submit', async event => { event.preventDefault(); const task = input.value.trim(); if (!task || sendButton.disabled) return; if (task.length > 8000) { addMessage('Please keep your message under 8,000 characters.', 'nyra'); return; } history.push({ role: 'user', content: task }); history = history.slice(-MAX_HISTORY); saveHistory(); addMessage('You: ' + task, 'user'); input.value = ''; sendButton.disabled = true; const thinking = document.createElement('div'); thinking.className = 'message nyra'; thinking.textContent = 'Nyra is thinking...'; messagesEl.appendChild(thinking); messagesEl.scrollTop = messagesEl.scrollHeight; try { const response = await fetch('/nyra/api/chat', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message: task, history: history.slice(0, -1) }) }); const data = await response.json(); if (!response.ok) { throw new Error(data.error || 'Request failed'); } if (typeof data.reply !== 'string' || !data.reply.trim()) { throw new Error('Nyra returned an empty response.'); } thinking.remove(); history.push({ role: 'assistant', content: data.reply }); history = history.slice(-MAX_HISTORY); saveHistory(); addMessage('Nyra: ' + data.reply, 'nyra'); } catch (error) { thinking.remove(); // Remove the unanswered user message from saved context. history.pop(); saveHistory(); addMessage( 'Sorry, I could not connect. ' + error.message, 'nyra' ); } finally { sendButton.disabled = false; input.focus(); } });
